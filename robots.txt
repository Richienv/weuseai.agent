# useful to crawlers; admin tooling under /admin is never indexable.
# Everything else — landing, use-cases, faq, checkout, persona pages,
# and the legal pages — is open on purpose.

Disallow: /onboarding.html
Disallow: /welcome
Disallow: /welcome.html
Disallow: /chat
Disallow: /chat.html
Disallow: /checkout-new
Disallow: /checkout-new.html
Disallow: /admin
